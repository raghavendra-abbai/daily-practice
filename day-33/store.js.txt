import { legacy_createStore as createStore } from "redux";

const initialState = {
  name: "Raghav",
  balance: 0,
  mobile: "9876543210"
};

function reducer(state = initialState, action) {

  switch (action.type) {

    // Deposit
    case "deposit":

      return {
        ...state,
        balance: state.balance + action.payload
      };

    // Withdraw
    case "withdraw":

      return {
        ...state,
        balance: state.balance - action.payload
      };

    // Update Name
    case "updateName":

      return {
        ...state,
        name: action.payload
      };

    // Update Mobile
    case "updateMobile":

      return {
        ...state,
        mobile: action.payload
      };

    default:
      return state;
  }
}

const store = createStore(reducer);

export default store;
