
import { useState } from "react";

function App(){
  let [counter, setcounter] = useState(0);
  let stock = 50;

  return(
    <div className="wrap">
      
      <button className="minus" onClick={()=>{
        if(counter > 0){
          setcounter(counter - 1)
        }
      }}>-</button>

      <p>{counter}</p>

      <button className="plus" onClick={()=>{
        if(counter < stock){
          setcounter(counter + 1)
        }
      }}>+</button>  

    </div>
  )
}

export default App;
