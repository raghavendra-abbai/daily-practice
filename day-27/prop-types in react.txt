import React from "react";
import PropTypes from "prop-types";

function User(props){
  return(
    <>
      <h1>This is an user component </h1>
      <h2>Hello:{props.name}</h2>
      <h2>Age:{props.age}</h2>
    </>
  )
}

User.propTypes = {
  name: PropTypes.string.isRequired,
  age: PropTypes.number.isRequired
}

function App(){
  return(
    <User name={"Raghav"} age={20}/>
  )
}

export default App;
