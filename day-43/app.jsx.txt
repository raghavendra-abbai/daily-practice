import { useDispatch, useSelector } from "react-redux";
import { fetchUser } from "./userSlice";

function App() {
  const dispatch = useDispatch();

  const { user, status, error } = useSelector(
    (state) => state.user
  );

  return (
    <div>
      <h1>Redux Toolkit AsyncThunk Example</h1>

      <button onClick={() => dispatch(fetchUser(1))}>
        Fetch User
      </button>

      {status === "loading" && <p>Loading...</p>}

      {status === "failed" && <p>Error: {error}</p>}

      {status === "succeeded" && (
        <div>
          <h2>{user.name}</h2>
          <p>{user.email}</p>
          <p>{user.phone}</p>
        </div>
      )}
    </div>
  );
}

export default App;
