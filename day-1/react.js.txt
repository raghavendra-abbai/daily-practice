import React from "react";

class App extends React.Component {
  constructor(props) {
    super(props);
    this.state = {
      product: "hp laptop",
      price: 80000,
      inputValue: "", 
    };
  }

  handleInputChange = (e) => {
    this.setState({ inputValue: e.target.value });
  };

  updateCost = () => {
    this.setState({ price: Number(this.state.inputValue) });
  };

  render() {
    return (
      <>
        <h1>product: {this.state.product}</h1>
        <p>price: {this.state.price}</p>

        <input
          type="number"
          value={this.state.inputValue}
          onChange={this.handleInputChange}
        />
        <button onClick={this.updateCost}>update cost</button>
      </>
    );
  }
}

 export default App;
