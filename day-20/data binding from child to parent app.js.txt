
import { useState } from "react";
import Search from "./search";

function App() {

 const [Todos, setTodos] = useState([
        
  'input-1',
   'input-2'

 ])
   
 let addnew=(task)=>{
  setTodos([...Todos, task])

 }

  return (
    <>
<Search addTodo={addnew} />
          <ul>
            {
            Todos.map((t)=>
              <li>{t}</li>  
            )
          }
          </ul>
          </>

   )
  }
  export default App;
